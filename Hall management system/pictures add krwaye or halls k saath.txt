class CustomerDashboard extends StatefulWidget {
  final String email;
  final String? profileImage;
  final String? name;

  CustomerDashboard({
    required this.email,
    this.profileImage,
    this.name,
  });

  @override
  _CustomerDashboardState createState() => _CustomerDashboardState();
}

class _CustomerDashboardState extends State<CustomerDashboard> {
  String? updatedProfileImage;
  String displayName = "";
  final String baseUrl = "http://172.16.110.71:5000";

  @override
  void initState() {
    super.initState();
    updatedProfileImage = widget.profileImage;

    if (widget.name != null && widget.name!.isNotEmpty) {
      displayName = widget.name!;
    } else {
      fetchCustomerName();
    }
  }

  Future<void> fetchCustomerName() async {
    try {
      final response = await http.get(Uri.parse('$baseUrl/get_user_name/${widget.email}'));
      if (response.statusCode == 200) {
        final data = jsonDecode(response.body);
        setState(() {
          displayName = data['name'] ?? formatNameFromEmail(widget.email);
        });
      } else {
        setState(() {
          displayName = formatNameFromEmail(widget.email);
        });
      }
    } catch (_) {
      setState(() {
        displayName = formatNameFromEmail(widget.email);
      });
    }
  }

  String formatNameFromEmail(String email) {
    try {
      final parts = email.split('@')[0].split('.');
      return parts.map((e) => e[0].toUpperCase() + e.substring(1)).join(' ');
    } catch (_) {
      return "User";
    }
  }

  void confirmLogout(BuildContext context) {
    showDialog(
      context: context,
      builder: (ctx) => AlertDialog(
        shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(16)),
        title: Text("Logout", style: GoogleFonts.poppins(fontWeight: FontWeight.bold)),
        content: Text("Are you sure you want to logout?", style: GoogleFonts.poppins()),
        actions: [
          TextButton(
            child: Text("Cancel", style: GoogleFonts.poppins()),
            onPressed: () => Navigator.of(ctx).pop(),
          ),
          ElevatedButton.icon(
            icon: Icon(Icons.logout),
            label: Text("Logout", style: GoogleFonts.poppins()),
            style: ElevatedButton.styleFrom(
              backgroundColor: Colors.redAccent,
              foregroundColor: Colors.white,
            ),
            onPressed: () {
              Navigator.of(ctx).pop();
              Navigator.pushNamedAndRemoveUntil(context, '/', (route) => false);
            },
          ),
        ],
      ),
    );
  }

  void goToProfile(BuildContext context) async {
    final result = await Navigator.push(
      context,
      MaterialPageRoute(
        builder: (_) => CustomerProfilePage(
          email: widget.email,
          name: displayName,
          profileImage: updatedProfileImage,
        ),
      ),
    );

    if (result != null && mounted) {
      try {
        final updateRes = await http.put(
          Uri.parse('$baseUrl/update_profile_image'),
          headers: {"Content-Type": "application/json"},
          body: jsonEncode({
            "email": widget.email,
            "profile_image": result,
          }),
        );

        if (updateRes.statusCode == 200) {
          setState(() {
            updatedProfileImage = '$baseUrl$result';
          });
        }
      } catch (e) {
        print("❌ Error updating profile image: $e");
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    final String customerName = displayName;
    final String customerContact = "03123456789"; // optional to make dynamic

    return Scaffold(
      backgroundColor: Colors.pink[50],
      appBar: AppBar(
        title: Text("🎉 Customer Dashboard", style: GoogleFonts.poppins(fontWeight: FontWeight.w600)),
        centerTitle: true,
        backgroundColor: Colors.pink[300],
        elevation: 10,
        shape: RoundedRectangleBorder(
          borderRadius: BorderRadius.vertical(bottom: Radius.circular(16)),
        ),
        actions: [
          IconButton(

class AuthScreen extends StatefulWidget {
  final bool initialLogin;

  AuthScreen({this.initialLogin = true});

  @override
  _AuthScreenState createState() => _AuthScreenState();
}

class _AuthScreenState extends State<AuthScreen> {
  late bool isLogin;
  final String baseUrl = "http://172.16.110.71:5000"; // Added baseUrl here

  final nameController = TextEditingController();
  final emailController = TextEditingController();
  final phoneController = TextEditingController();
  final passwordController = TextEditingController();
  final profileImageController = TextEditingController();

  @override
  void initState() {
    super.initState();
    isLogin = widget.initialLogin;
  }

  void toggleMode() => setState(() => isLogin = !isLogin);

  Future<void> submit() async {
    String url = isLogin ? '$baseUrl/login' : '$baseUrl/register';

    Map<String, dynamic> bodyData = isLogin
        ? {
            "email": emailController.text.trim(),
            "password": passwordController.text,
          }
        : {
            "name": nameController.text.trim(),
            "email": emailController.text.trim(),
            "phone": phoneController.text.trim(),
            "password": passwordController.text,
            "profile_image": profileImageController.text.trim(),
          };

    try {
      final response = await http.post(
        Uri.parse(url),
        headers: {"Content-Type": "application/json"},
        body: jsonEncode(bodyData),
      );

      final data = jsonDecode(response.body);

      if (response.statusCode == 200 || response.statusCode == 201) {
        if (isLogin) {
          final String role = data['role']?.toString() ?? '';
          final String email = data['email']?.toString() ?? '';
          final String profileImage = data['profile_image']?.toString() ?? '';
          final String name = data['name']?.toString() ?? '';

          if (role.isEmpty) {
            showError("❌ Invalid response from server.");
            return;
          }

          if (role == 'admin') {
            if (mounted) {
              Navigator.pushReplacement(
                context,
                MaterialPageRoute(builder: (_) => AdminDashboard()),
              );
            }
          } else if (role == 'customer') {
            if (mounted) {
              Navigator.pushReplacement(
                context,
                MaterialPageRoute(
                  builder: (_) => CustomerDashboard(
                    email: email,
                    profileImage: profileImage,
                    name: name,
                  ),
                ),
              );
            }
          } else if (role == 'client') {
            if (mounted) {
              Navigator.pushReplacement(
                context,
                MaterialPageRoute(
                  builder: (_) => ClientDashboard(
                    clientEmail: email,
                    name: name,
                  ),
                ),
              );
            }
          } else {
            showError("❌ Unknown role. Please contact support.");
          }
        } else {
          showSuccess(data['message'] ?? 'Registration successful. Please log in.');
        }
      } else {
        showError(data['message'] ?? data['error'] ?? '❌ Something went wrong');
      }
    } catch (e) {
      showError("❌ Network error: ${e.toString()}");
    }
  }

  void showError(String msg) {
    if (mounted) {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text(msg), backgroundColor: Colors.red),
      );
    }
  }

  void showSuccess(String msg) {
    if (mounted) {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text(msg), backgroundColor: Colors.green),
      );
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: Stack(
        children: [
          Container(
            decoration: BoxDecoration(
              image: DecorationImage(
                image: AssetImage("assets/elegant-white-blank-wedding-invitation-surrounded-delicate-pink-roses-artistic-rendering-creates-inviting-template-floral-357362509.webp"),
                fit: BoxFit.cover,
              ),

              child: Column(
                mainAxisSize: MainAxisSize.min,
                children: [
                  Text(
                    isLogin ? "Login" : "Register",
                    style: TextStyle(fontSize: 34, fontWeight: FontWeight.bold, color: Colors.white),

                  if (!isLogin)
                    TextField(

                      decoration: inputDecoration('Name'),
                    ),
                  SizedBox(height: 12),
                  TextField(

                    style: TextStyle(color: Colors.white),

class CalendarBookingPage extends StatefulWidget {
  final String email;
  final int hallId;
  final String hallName;
  final String customerName;
  final String customerContact;

  CalendarBookingPage({
    required this.email,
    required this.hallId,
    required this.hallName,
    required this.customerName,
    required this.customerContact,
  });

  @override
  _CalendarBookingPageState createState() => _CalendarBookingPageState();
}

class _CalendarBookingPageState extends State<CalendarBookingPage> {
  DateTime focusedDate = DateTime.now();
  DateTime? selectedDate;
  List<String> availableDates = [];

  final String baseUrl = "http://172.16.110.71:5000"; // 🔗 Your backend

  @override
  void initState() {
    super.initState();
    fetchAvailability();
  }

  Future<void> fetchAvailability() async {
    final res = await http.get(Uri.parse('$baseUrl/hall/${widget.hallId}/availability'));
    if (res.statusCode == 200) {
      final data = jsonDecode(res.body);
      setState(() {
        availableDates = List<String>.from(data['available_dates']); // Format: yyyy-MM-dd
      });
    } else {
      print("❌ Failed to fetch availability");
    }
  }

  bool isAvailable(DateTime day) {
    String formatted = DateFormat('yyyy-MM-dd').format(day);
    return availableDates.contains(formatted);
  }

  Future<void> bookSelectedDate() async {
    if (selectedDate == null) {
      ScaffoldMessenger.of(context).showSnackBar(SnackBar(
        content: Text("❌ Please select a date first."),
        backgroundColor: Colors.red,
      ));
      return;
    }

    String formattedDate = DateFormat('yyyy-MM-dd').format(selectedDate!);

    final Uri url = Uri.parse(
      '$baseUrl/book?email=${Uri.encodeComponent(widget.email)}'
      '&hall_id=${widget.hallId}'
      '&booking_date=$formattedDate'
      '&customer_name=${Uri.encodeComponent(widget.customerName)}'
      '&customer_contact_number=${Uri.encodeComponent(widget.customerContact)}',
    );

    try {
      final response = await http.get(url);
      final data = jsonDecode(response.body);

      ScaffoldMessenger.of(context).showSnackBar(SnackBar(
        content: Text(data['message'] ?? '❌ Booking failed'),
        backgroundColor: response.statusCode == 201 ? Colors.green : Colors.red,
      ));

      if (response.statusCode == 201) {
        await fetchAvailability(); // 🔁 Refresh calendar
        Navigator.pop(context);
      }
    } catch (e) {
      ScaffoldMessenger.of(context).showSnackBar(SnackBar(
        content: Text("❌ Error: $e"),
        backgroundColor: Colors.red,
      ));
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: Text("📅 Book ${widget.hallName}"),
        backgroundColor: Colors.pink[300],
        centerTitle: true,
      ),
      body: Column(
        children: [
          TableCalendar(
            firstDay: DateTime.now().subtract(Duration(days: 365)),
            lastDay: DateTime.now().add(Duration(days: 36500)),
            focusedDay: focusedDate,
            selectedDayPredicate: (day) => isSameDay(selectedDate, day),
            onDaySelected: (selected, focused) {
              if (isAvailable(selected)) {
                setState(() {
                  selectedDate = selected;
                  focusedDate = focused;
                });
              } else {
                ScaffoldMessenger.of(context).showSnackBar(SnackBar(
                  content: Text("❌ This date is already booked."),
                  backgroundColor: Colors.red,
                ));
              }
            },
            calendarStyle: CalendarStyle(
              isTodayHighlighted: true,
              selectedDecoration: BoxDecoration(
                color: Colors.indigo,
                shape: BoxShape.circle,
              ),
              todayDecoration: BoxDecoration(
                color: Colors.orange,
                shape: BoxShape.circle,
              ),
              disabledTextStyle: TextStyle(color: Colors.grey),
            ),
            enabledDayPredicate: (day) => isAvailable(day),
          ),
          SizedBox(height: 20),
          ElevatedButton.icon(
            onPressed: bookSelectedDate,
            icon: Icon(Icons.done),
            label: Text("Confirm Booking"),
            style: ElevatedButton.styleFrom(
              backgroundColor: Colors.green,
              foregroundColor: Colors.white,
              padding: EdgeInsets.symmetric(horizontal: 24, vertical: 14),
              shape: RoundedRectangleBorder(
                borderRadius: BorderRadius.circular(12),
              ),
            ),
          ),
        ],
      ),
    );
  }
}

class _MyBookingsPageState extends State<MyBookingsPage> {
  final String baseUrl = "http://172.16.110.71:5000";

  Future<List<dynamic>> fetchBookings() async {
    try {
      final url = '$baseUrl/my_bookings?email=${widget.email}';
      print("📡 Fetching bookings from: $url");
      final response = await http.get(Uri.parse(url));
      print("✅ Status Code: ${response.statusCode}");
      print("📦 Response Body: ${response.body}");

      if (response.statusCode == 200) {
        final Map<String, dynamic> data = jsonDecode(response.body);
        print("✅ Parsed bookings: ${data['bookings']}");
        return data['bookings'] ?? [];
      } else {
        print("⚠️ Server returned error status.");
        return [];
      }
    } catch (e) {
      print("❌ Error fetching bookings: $e");
      return [];
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: Text("My Bookings", style: GoogleFonts.poppins()),
        backgroundColor: Colors.deepPurple,
        foregroundColor: Colors.white,
      ),
      body: FutureBuilder<List<dynamic>>(
        future: fetchBookings(),
        builder: (context, snapshot) {
          if (snapshot.connectionState == ConnectionState.waiting) {
            return Center(child: CircularProgressIndicator(color: Colors.deepPurple));
          } else if (snapshot.hasError) {
            return Center(child: Text('Error: ${snapshot.error}', style: GoogleFonts.poppins()));
          } else {
            final bookings = snapshot.data ?? [];

            if (bookings.isEmpty) {
              return Center(
                child: Text("No bookings found.", style: GoogleFonts.poppins(fontSize: 18)),
              );
            }

            return ListView.builder(
              itemCount: bookings.length,
              itemBuilder: (context, index) {
                final booking = bookings[index];
                return Card(
                  margin: EdgeInsets.all(12),
                  elevation: 4,
                  shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(12)),
                  child: ListTile(
                    title: Text(booking['hall_name'] ?? "Unknown Hall", style: GoogleFonts.poppins(fontWeight: FontWeight.bold)),
                    subtitle: Text("Date: ${booking['date']}\nStatus: ${booking['status']}", style: GoogleFonts.poppins()),
                    leading: Icon(Icons.event, color: Colors.deepPurple),
                  ),
                );
              },
            );
          }
        },
      ),
    );
  }

class AdminDashboard extends StatefulWidget {
  @override
  _AdminDashboardState createState() => _AdminDashboardState();
}

class _AdminDashboardState extends State<AdminDashboard> {
  late Future<List<dynamic>> bookingsFuture;
  List<dynamic> allBookings = [];
  List<dynamic> filteredBookings = [];
  final String baseUrl = "http://172.16.110.71:5000"; // Added baseUrl here

  DateTime? fromDate;
  DateTime? toDate;
  String searchQuery = "";
  String selectedStatus = "All";

  final Color gunMetal = Color(0xFF2A3439);
  final Color lightGrey = Color(0xFFF2F2F2);
  final ScrollController _scrollController = ScrollController();

  @override
  void initState() {
    super.initState();
    bookingsFuture = fetchBookings();
  }

  Future<List<dynamic>> fetchBookings() async {
    try {
      final res = await http.get(Uri.parse('$baseUrl/admin/bookings'));
      print("📥 Status: ${res.statusCode}");
      print("📥 Response: ${res.body}");

      if (res.statusCode == 200) {
        final data = jsonDecode(res.body);
        setState(() {
          allBookings = data;
          filteredBookings = data;
        });
        return data;
      } else {
        throw Exception("Status Code: ${res.statusCode}");
      }
    } catch (e) {
      print("❌ Error fetching bookings: $e");
      return [];
    }
  }

  void _applyFilters() {
    List<dynamic> filtered = allBookings;

    if (fromDate != null && toDate != null) {
      filtered = filtered.where((b) {
        final dateStr = b['booking_date'] ?? '';
        try {
          final bookingDate = DateTime.parse(dateStr);
          return bookingDate.isAfter(fromDate!.subtract(Duration(days: 1))) &&
              bookingDate.isBefore(toDate!.add(Duration(days: 1)));
        } catch (_) {
          return false;
        }
      }).toList();
    }

    if (selectedStatus != "All") {
      filtered = filtered.where((b) {
        final status = (b['status'] ?? '').toString().toLowerCase();
        return status == selectedStatus.toLowerCase();
      }).toList();
    }

    if (searchQuery.isNotEmpty) {
      filtered = filtered.where((b) {
        final hall = (b['hall_name'] ?? '').toString().toLowerCase();
        return hall.contains(searchQuery.toLowerCase());
      }).toList();
    }

    setState(() {
      filteredBookings = filtered;
    });
  }

  void _selectStartDate() async {
    final picked = await showDatePicker(
      context: context,
      initialDate: fromDate ?? DateTime.now(),
      firstDate: DateTime(2023),
      lastDate: DateTime(2100),
    );
    if (picked != null) setState(() => fromDate = picked);
  }

  void _selectEndDate() async {
    final picked = await showDatePicker(
      context: context,
      initialDate: toDate ?? DateTime.now(),
      firstDate: DateTime(2023),
      lastDate: DateTime(2100),
    );
    if (picked != null) setState(() => toDate = picked);
  }

    setState(() {
      fromDate = null;
      toDate = null;
      selectedStatus = "All";
      searchQuery = "";
      filteredBookings = allBookings;
    });
  }

      context: context,
      builder: (BuildContext dialogContext) {
        return AlertDialog(
          backgroundColor: lightGrey,
          title: Text("Confirm $capitalized", style: TextStyle(color: gunMetal)),
          content: Text("Are you sure you want to $action this booking?", style: TextStyle(color: gunMetal)),
          actions: [

class ClientDashboard extends StatefulWidget {
  final String clientEmail;
  final String name;

  ClientDashboard({
    required this.clientEmail,
    required this.name,
  });

  @override
  _ClientDashboardState createState() => _ClientDashboardState();
}

class _ClientDashboardState extends State<ClientDashboard> {
  final String baseUrl = "http://172.16.110.71:5000";
  List<dynamic> clientHalls = [];
  List<dynamic> clientBookings = [];
  bool isLoading = true;

  @override
  void initState() {
    super.initState();
    fetchClientData();
  }

  Future<void> fetchClientData() async {
    setState(() => isLoading = true);
    await Future.wait([
      fetchClientHalls(),
      fetchClientBookings(),
    ]);
    setState(() => isLoading = false);
  }

  Future<void> fetchClientHalls() async {
    final response = await http.get(
      Uri.parse('$baseUrl/client/halls/${widget.clientEmail}'),
    );

    if (response.statusCode == 200) {
      setState(() {
        clientHalls = json.decode(response.body)['client_halls'];
      });
    } else {
      print("❌ Failed to load halls.");
    }
  }

  Future<void> fetchClientBookings() async {
    final response = await http.get(
      Uri.parse('$baseUrl/client/bookings/${widget.clientEmail}'),
    );

    if (response.statusCode == 200) {
      setState(() {
        clientBookings = json.decode(response.body)['bookings'];
      });
    } else {
      print("❌ Failed to load bookings.");
    }
  }

  String formatDate(String dateStr) {
    try {
      final dt = DateTime.parse(dateStr);
      return DateFormat('dd MMM yyyy').format(dt);
    } catch (e) {
      return dateStr;
    }
  }

  void showConfirmationDialog(int bookingId, String status, String hallName, String bookingDate) {
    final isApproval = status == 'confirmation';
    final warningText = isApproval
        ? "Approving this booking will automatically reject all other bookings for the same hall and date."
        : "Are you sure you want to cancel this booking?";

    showDialog(
      context: context,
      builder: (ctx) => AlertDialog(
        title: Text(isApproval ? "Approve Booking?" : "Cancel Booking?"),
        content: Text(warningText),
        actions: [
          TextButton(
            onPressed: () => Navigator.of(ctx).pop(),
            child: Text("Cancel"),
          ),
          ElevatedButton(
            onPressed: () {
              Navigator.of(ctx).pop();
              updateBookingStatus(bookingId, status);
            },
            style: ElevatedButton.styleFrom(
              backgroundColor: isApproval ? Colors.green : Colors.red,
              foregroundColor: Colors.white,
            ),
            child: Text(isApproval ? "Approve" : "Cancel"),
          ),
        ],
      ),
    );
  }

  Future<void> updateBookingStatus(int bookingId, String status) async {
    final url = Uri.parse('$baseUrl/update_booking/$bookingId');

    final response = await http.put(
      url,
      headers: {"Content-Type": "application/json"},
      body: jsonEncode({
        "status": status,
        "email": widget.clientEmail,
        "role": "client",
      }),
    );

    if (response.statusCode == 200) {
      // Find all bookings for the same hall and date
      final approvedBooking = clientBookings.firstWhere(
        (booking) => booking['booking_id'] == bookingId,
        orElse: () => null,
      );

      if (approvedBooking != null && status == 'confirmation') {
        final hallName = approvedBooking['hall_name'];
        final bookingDate = approvedBooking['booking_date'];

        // Update local state to reflect rejected bookings
        setState(() {
          clientBookings = clientBookings.map((booking) {
            if (booking['hall_name'] == hallName && 
                booking['booking_date'] == bookingDate && 
                booking['booking_id'] != bookingId &&
                (booking['status'] == 'pending' || booking['status'] == 'confirmation')) {
              return {...booking, 'status': 'rejected'};
            }
            return booking;
          }).toList();
        });
      }

      // Update the status of the current booking
      setState(() {
        clientBookings = clientBookings.map((booking) {
          if (booking['booking_id'] == bookingId) {
            return {...booking, 'status': status};
          }
          return booking;
        }).toList();
      });

      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text("✅ Booking ${status == 'confirmation' ? 'approved' : 'canceled'} successfully.")),
      );
    } else {
      print("❌ Error Response: ${response.body}");
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text("❌ Failed to update booking.")),
      );
    }
  }

  Color getStatusColor(String status) {
    switch (status) {
      case 'confirmation':
        return Colors.green;
      case 'canceled':
      case 'rejected':
        return Colors.red;
      case 'pending':
      default:
        return Colors.orange;
    }
  }

  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: Text("Welcome, ${widget.name}"),
        backgroundColor: Colors.pink[300],
        actions: [
          IconButton(
            icon: Icon(Icons.refresh),
            onPressed: fetchClientData,
          )
        ],
      ),
      body: isLoading
          ? Center(child: CircularProgressIndicator())
          : RefreshIndicator(
              onRefresh: fetchClientData,
              child: SingleChildScrollView(
                physics: AlwaysScrollableScrollPhysics(),
                padding: EdgeInsets.all(12),
                child: Column(
                  crossAxisAlignment: CrossAxisAlignment.start,
                  children: [
                    Text("🏠 Your Halls",
                        style: TextStyle(
                          fontSize: 18,
                          fontWeight: FontWeight.bold,
                          color: Colors.pink[900],
                        )),
                    SizedBox(height: 8),
                    ...clientHalls.map((hall) => Card(
                          child: ListTile(
                            title: Text(hall['hall_name']),
                            subtitle: Text("📍 ${hall['location']}"),
                          ),
                        )),
                    SizedBox(height: 20),
                    Text("📋 Booking Requests",
                        style: TextStyle(
                          fontSize: 18,
                          fontWeight: FontWeight.bold,
                          color: Colors.pink[900],
                        )),
                    SizedBox(height: 8),
                    ...clientBookings.map((booking) => Card(
                          elevation: 3,
                          margin: EdgeInsets.symmetric(vertical: 8),
                          child: Padding(
                            padding: const EdgeInsets.all(12.0),
                            child: Column(
                              crossAxisAlignment: CrossAxisAlignment.start,

                                Text("🏠 Hall: ${booking['hall_name']}",
                                    style: TextStyle(

                                Text("📞 Contact: ${booking['customer_contact_number']}"),

                                  style: TextStyle(
                                    color: getStatusColor(booking['status']),
                                    fontWeight: FontWeight.bold,
                                  ),
                                ),
                                SizedBox(height: 10),
                                if (booking['status'] == 'pending')
                                  Row(

                            ),
                          ),
                        )),
                  ],
                ),
              ),
            ),
    );
  }

def register():
    data = request.get_json()

    existing_user = Customer.query.filter_by(email=email).first()
    if existing_user:
        return jsonify({"error": "User already exists"}), 409

    new_user = Customer(
        name=name,

        phone=phone,
        password=password,
        profile_image=profile_image,
        role='customer'
    )

    db.session.add(new_user)
    db.session.commit()

    log_action(email, 'customer', 'Registered', 'New customer registered')

    return jsonify({"message": "User registered successfully."}), 201

        password = data.get('password', '').strip()

        # Query for matching customer (including client)
        user = Customer.query.filter(

@app.route('/upload_profile_image', methods=['POST'])
def upload_profile_image():
    if 'image' not in request.files or 'email' not in request.form:
        return jsonify({"error": "Image or email missing"}), 400

    file = request.files['image']
    email = request.form['email'].strip().lower()

    if file.filename == '':
        return jsonify({"error": "No file selected"}), 400

    filename = secure_filename(file.filename)
    filepath = os.path.join(app.config['UPLOAD_FOLDER'], filename)
    file.save(filepath)

    user = Customer.query.filter_by(email=email).first()
    if user:
        user.profile_image = f"/uploads/{filename}"
        db.session.commit()
        return jsonify({"status": "success", "profile_image": user.profile_image})

    return jsonify({"error": "User not found"}), 404

@app.route('/book', methods=['GET'])  # Forced GET booking
def book():
    email = request.args.get('email')
    hall_id = request.args.get('hall_id')
    booking_date = request.args.get('booking_date')  # Format: YYYY-MM-DD
    customer_name = request.args.get('customer_name')
    customer_contact_number = request.args.get('customer_contact_number')

    if not all([email, hall_id, booking_date, customer_name, customer_contact_number]):
        return jsonify({"error": "❌ Missing fields"}), 400

    try:
        booking_date_obj = datetime.strptime(booking_date, '%Y-%m-%d').date()
        hall_id_int = int(hall_id)
        
        print(f"📅 Booking attempt => Hall: {hall_id_int}, Date: {booking_date_obj}")

        # Check if booking already exists by this customer on same hall and date
        duplicate = db.session.execute(
            text("""
                SELECT 1 FROM bookings
                WHERE customer_email = :email AND hall_id = :hall_id AND booking_date = :booking_date
            """),
            {"email": email, "hall_id": hall_id_int, "booking_date": booking_date_obj}
        ).fetchone()

        if duplicate:
            return jsonify({"error": "❌ Booking already requested for this date by this customer."}), 409

        # Check availability
        record = db.session.execute(
            text("""
                SELECT * FROM hall_availability
                WHERE hall_id = :hall_id AND available_date = :booking_date AND is_booked = 0
            """),
            {"hall_id": hall_id_int, "booking_date": booking_date_obj}
        ).fetchone()

        if not record:
            return jsonify({"error": "❌ Date not found for this hall or already booked"}), 409

        # Insert booking
        new_booking = Booking(
            customer_email=email,
            hall_id=hall_id_int,
            booking_date=booking_date_obj,
            status="pending",
            customer_name=customer_name,
            customer_contact_number=customer_contact_number
        )
        db.session.add(new_booking)

        db.session.commit()

        # Log action
        log_action(email, 'customer', "Booking Request", f"Hall ID: {hall_id}, Date: {booking_date}")

        return jsonify({"message": "✅ Booking requested successfully."}), 201

    except Exception as e:
        return jsonify({"error": f"❌ Server error: {str(e)}"}), 500

@app.route('/approve_booking', methods=['GET'])
def approve_booking():
    booking_id = request.args.get('booking_id')
    new_status = request.args.get('status')  # 'approved_by_admin' or 'approved_by_client'
    approver_email = request.args.get('email')  # Email of admin/client approving

    # Fetch the target booking with hall details
    booking = db.session.query(Booking, MarriageHall).join(
        MarriageHall, Booking.hall_id == MarriageHall.hall_id
    ).filter(Booking.hall_booking_id == booking_id).first()

    if not booking:
        return jsonify({'error': 'Booking not found'}), 404

    booking_obj, hall = booking
    hall_id = booking_obj.hall_id
    booking_date = booking_obj.booking_date

    # If approving, reject all other bookings for the same hall/date
    if new_status in ['approved_by_admin', 'approved_by_client']:
        try:
            # Reject all other pending bookings
            db.session.execute(
                text("""
                    UPDATE bookings
                    SET status = 'rejected'
                    WHERE hall_id = :hall_id 
                    AND booking_date = :booking_date 
                    AND hall_booking_id != :id
                    AND status = 'pending'
                """),
                {'hall_id': hall_id, 'booking_date': booking_date, 'id': booking_id}
            )

            # Mark the hall as booked on that date
            db.session.execute(
                text("""
                    UPDATE hall_availability
                    SET is_booked = 1
                    WHERE hall_id = :hall_id AND available_date = :booking_date
                """),
                {'hall_id': hall_id, 'booking_date': booking_date}
            )

            # Update the approved booking
            booking_obj.status = new_status
            db.session.commit()

            # Log all actions
            log_action(approver_email, 'admin' if 'admin' in new_status else 'client', 
                      f"Approved booking {booking_id}", 
                      f"Hall: {hall.hall_name}, Date: {booking_date}")
            
            # Log auto-rejections
            rejected_bookings = Booking.query.filter(
                Booking.hall_id == hall_id,
                Booking.booking_date == booking_date,
                Booking.hall_booking_id != booking_id,
                Booking.status == 'rejected'
            ).all()
            
            for rb in rejected_bookings:
                log_action(rb.customer_email, 'system', "Booking auto-rejected", 
                          f"Due to another booking being approved for {hall.hall_name} on {booking_date}")

            return jsonify({
                'message': 'Booking approved successfully',
                'rejected_count': len(rejected_bookings)
            })

        except Exception as e:
            db.session.rollback()
            return jsonify({'error': str(e)}), 500
